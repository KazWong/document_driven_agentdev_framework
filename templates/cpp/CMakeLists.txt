cmake_minimum_required(VERSION 3.20)
project(__MODULE_NAME__ VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(${PROJECT_NAME} src/library.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC include)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${PROJECT_NAME} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${PROJECT_NAME} PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(test_${PROJECT_NAME} test/test_library.cpp)
target_link_libraries(test_${PROJECT_NAME} PRIVATE ${PROJECT_NAME})
target_compile_options(test_${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_${PROJECT_NAME} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test_${PROJECT_NAME} PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME unit COMMAND test_${PROJECT_NAME})
